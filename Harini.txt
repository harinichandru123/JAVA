class Student {
    String name;
    int rollNo;
    double marks;
    void display() {
        System.out.println("Name:" + name);
        System.out.println("Roll No:" + rollNo);
        System.out.println("Marks :" + marks);
    }
    public static void main(String[] args) {
        Student s1 = new Student();
        s1.name = "Kaviya";
        s1.rollNo = 70;
        s1.marks = 85;
        s1.display();
    }
}
